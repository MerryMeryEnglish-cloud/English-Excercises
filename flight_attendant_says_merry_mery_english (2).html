// Estado para llevar el control de la rotación de voces let voiceGenderToggle = false; // false = femenina, true = masculina function playBritishAudio(textElementId) { const text = document.getElementById(textElementId).innerText; const utterance = new SpeechSynthesisUtterance(text); // Obtener las voces disponibles en el navegador const voices = window.speechSynthesis.getVoices(); // Filtrar voces en inglés británico (en-GB) const britishVoices = voices.filter(voice => voice.lang === 'en-GB' || voice.lang === 'en_GB'); if (britishVoices.length > 0) { // Intentar separar por género aproximando por nombre (ej. 'Female', 'Male', 'George', 'Hazel', etc.) const femaleVoices = britishVoices.filter(v => v.name.toLowerCase().includes('female') || v.name.includes('Hazel') || v.name.includes('Libby') || v.name.includes('Maisie')); const maleVoices = britishVoices.filter(v => v.name.toLowerCase().includes('male') || v.name.includes('George') || v.name.includes('Ryan') || v.name.includes('Oliver')); let selectedVoice; if (!voiceGenderToggle && femaleVoices.length > 0) { selectedVoice = femaleVoices[Math.floor(Math.random() * femaleVoices.length)]; voiceGenderToggle = true; // Siguiente turno será masculina } else if (voiceGenderToggle && maleVoices.length > 0) { selectedVoice = maleVoices[Math.floor(Math.random() * maleVoices.length)]; voiceGenderToggle = false; // Siguiente turno será femenina } else { // Si no se detectan etiquetas claras, toma cualquiera de la lista británica en rotación selectedVoice = britishVoices[Math.floor(Math.random() * britishVoices.length)]; } utterance.voice = selectedVoice; } utterance.rate = 0.95; // Velocidad ligeramente pausada ideal para práctica de aviación/listening window.speechSynthesis.speak(utterance); }
